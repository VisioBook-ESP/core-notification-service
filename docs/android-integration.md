# Android Integration Guide\n\nThis guide shows how to integrate Firebase push notifications in your Android app.\n\n## 📋 Prerequisites\n\n- Android Studio Giraffe+\n- Android SDK 21+\n- Gradle 8.0+\n- Firebase account with Android app configured\n- Google Play Services\n\n## 🚀 Installation\n\n### 1. Add Firebase Dependencies\n\n**build.gradle (Project level)**:\n```gradle\nbuildscript {\n    dependencies {\n        classpath 'com.google.gms:google-services:4.3.15'\n    }\n}\n```\n\n**build.gradle (App level)**:\n```gradle\nplugins {\n    id 'com.android.application'\n    id 'com.google.gms.google-services'\n}\n\ndependencies {\n    implementation 'com.google.firebase:firebase-messaging:23.2.1'\n    implementation 'com.google.firebase:firebase-core:21.1.1'\n    \n    // For HTTP requests to backend\n    implementation 'com.squareup.okhttp3:okhttp:4.11.0'\n    implementation 'com.google.code.gson:gson:2.10.1'\n}\n```\n\n### 2. Download google-services.json\n\n1. Go to [Firebase Console](https://console.firebase.google.com)\n2. Select your project → Project Settings\n3. Go to \"Your apps\" → Android section\n4. Download `google-services.json`\n5. Place in `app/` directory\n\n## 🔧 Implementation\n\n### Step 1: Create Messaging Service\n\n**VisioBookMessagingService.kt**:\n```kotlin\npackage com.visiobook.messaging\n\nimport android.app.NotificationChannel\nimport android.app.NotificationManager\nimport android.app.PendingIntent\nimport android.content.Context\nimport android.content.Intent\nimport android.content.SharedPreferences\nimport android.os.Build\nimport androidx.core.app.NotificationCompat\nimport com.google.firebase.messaging.FirebaseMessagingService\nimport com.google.firebase.messaging.RemoteMessage\nimport com.visiobook.MainActivity\nimport com.visiobook.R\nimport kotlinx.coroutines.Dispatchers\nimport kotlinx.coroutines.GlobalScope\nimport kotlinx.coroutines.launch\nimport okhttp3.MediaType.Companion.toMediaType\nimport okhttp3.OkHttpClient\nimport okhttp3.Request\nimport okhttp3.RequestBody.Companion.toRequestBody\nimport org.json.JSONObject\n\nclass VisioBookMessagingService : FirebaseMessagingService() {\n    \n    private val notificationManager: NotificationManager by lazy {\n        getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager\n    }\n    \n    private val preferences: SharedPreferences by lazy {\n        getSharedPreferences(\"visiobook_prefs\", Context.MODE_PRIVATE)\n    }\n    \n    companion object {\n        private const val CHANNEL_ID = \"visiobook_notifications\"\n        private const val TAG = \"VisioBookMessaging\"\n    }\n    \n    override fun onCreate() {\n        super.onCreate()\n        createNotificationChannel()\n    }\n    \n    override fun onNewToken(token: String) {\n        super.onNewToken(token)\n        android.util.Log.d(TAG, \"🔔 FCM Token: $token\")\n        \n        // Save token locally\n        preferences.edit().putString(\"fcm_token\", token).apply()\n        \n        // Register with backend\n        registerDeviceToken(token)\n    }\n    \n    private fun registerDeviceToken(token: String) {\n        val userId = preferences.getString(\"user_id\", null) ?: return\n        val jwtToken = preferences.getString(\"jwt_token\", null) ?: return\n        val deviceId = android.provider.Settings.Secure.getString(\n            contentResolver,\n            android.provider.Settings.Secure.ANDROID_ID\n        )\n        \n        GlobalScope.launch(Dispatchers.IO) {\n            try {\n                val client = OkHttpClient()\n                \n                val requestBody = JSONObject().apply {\n                    put(\"deviceId\", deviceId)\n                    put(\"platform\", \"android\")\n                    put(\"token\", token)\n                }.toString().toRequestBody(\"application/json\".toMediaType())\n                \n                val request = Request.Builder()\n                    .url(\"https://notifications-api.visiobook.com/api/v1/push/subscribe\")\n                    .addHeader(\"Authorization\", \"Bearer $jwtToken\")\n                    .post(requestBody)\n                    .build()\n                \n                val response = client.newCall(request).execute()\n                if (response.isSuccessful) {\n                    android.util.Log.d(TAG, \"✅ Device registered successfully\")\n                } else {\n                    android.util.Log.e(TAG, \"❌ Registration failed: ${response.code}\")\n                }\n                response.close()\n            } catch (e: Exception) {\n                android.util.Log.e(TAG, \"❌ Failed to register device: ${e.message}\", e)\n            }\n        }\n    }\n    \n    override fun onMessageReceived(remoteMessage: RemoteMessage) {\n        super.onMessageReceived(remoteMessage)\n        \n        android.util.Log.d(TAG, \"🔔 Message received from: ${remoteMessage.from}\")\n        \n        val title = remoteMessage.notification?.title ?: \"Visiobook\"\n        val body = remoteMessage.notification?.body ?: \"\"\n        val data = remoteMessage.data\n        \n        // Handle the notification\n        sendNotification(title, body, data)\n    }\n    \n    private fun sendNotification(\n        title: String,\n        body: String,\n        data: Map<String, String>\n    ) {\n        val intent = Intent(this, MainActivity::class.java).apply {\n            addFlags(Intent.FLAG_ACTIVITY_CLEAR_TOP)\n            data.forEach { (key, value) ->\n                putExtra(key, value)\n            }\n        }\n        \n        val pendingIntent = PendingIntent.getActivity(\n            this,\n            0,\n            intent,\n            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT\n        )\n        \n        val notificationBuilder = NotificationCompat.Builder(this, CHANNEL_ID)\n            .setContentTitle(title)\n            .setContentText(body)\n            .setAutoCancel(true)\n            .setContentIntent(pendingIntent)\n            .setSmallIcon(R.drawable.ic_notification)\n            .setPriority(NotificationCompat.PRIORITY_HIGH)\n        \n        val notificationId = System.currentTimeMillis().toInt()\n        notificationManager.notify(notificationId, notificationBuilder.build())\n        \n        android.util.Log.d(TAG, \"📲 Notification shown: $title\")\n    }\n    \n    private fun createNotificationChannel() {\n        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {\n            val channel = NotificationChannel(\n                CHANNEL_ID,\n                \"Visiobook Notifications\",\n                NotificationManager.IMPORTANCE_DEFAULT\n            ).apply {\n                description = \"Notifications from Visiobook\"\n                enableVibration(true)\n                enableLights(true)\n            }\n            notificationManager.createNotificationChannel(channel)\n        }\n    }\n}\n```\n\n### Step 2: Update AndroidManifest.xml\n\n```xml\n<?xml version=\"1.0\" encoding=\"utf-8\"?>\n<manifest xmlns:android=\"http://schemas.android.com/apk/res/android\">\n    \n    <!-- Permissions -->\n    <uses-permission android:name=\"android.permission.INTERNET\" />\n    <uses-permission android:name=\"android.permission.POST_NOTIFICATIONS\" />\n    \n    <application>\n        <!-- Firebase Messaging Service -->\n        <service\n            android:name=\".messaging.VisioBookMessagingService\"\n            android:exported=\"false\">\n            <intent-filter>\n                <action android:name=\"com.google.firebase.MESSAGING_EVENT\" />\n            </intent-filter>\n        </service>\n        \n        <!-- Main Activity -->\n        <activity\n            android:name=\".MainActivity\"\n            android:exported=\"true\">\n            <intent-filter>\n                <action android:name=\"android.intent.action.MAIN\" />\n                <category android:name=\"android.intent.category.LAUNCHER\" />\n            </intent-filter>\n        </activity>\n    </application>\n</manifest>\n```\n\n### Step 3: Initialize Firebase in MainActivity\n\n**MainActivity.kt**:\n```kotlin\npackage com.visiobook\n\nimport android.os.Bundle\nimport androidx.appcompat.app.AppCompatActivity\nimport com.google.firebase.messaging.FirebaseMessaging\n\nclass MainActivity : AppCompatActivity() {\n    \n    override fun onCreate(savedInstanceState: Bundle?) {\n        super.onCreate(savedInstanceState)\n        setContentView(R.layout.activity_main)\n        \n        // Get FCM token (for logging/debugging)\n        FirebaseMessaging.getInstance().token.addOnCompleteListener { task ->\n            if (task.isSuccessful) {\n                val token = task.result\n                android.util.Log.d(\"FCM\", \"Token: $token\")\n            }\n        }\n        \n        // Handle notification if app was opened from notification\n        intent.extras?.let { extras ->\n            handleNotificationData(extras)\n        }\n    }\n    \n    private fun handleNotificationData(data: Bundle) {\n        val messageId = data.getString(\"messageId\")\n        val action = data.getString(\"action\")\n        \n        when (action) {\n            \"open_message\" -> {\n                // Navigate to message screen\n            }\n            \"open_payment\" -> {\n                // Navigate to payment screen\n            }\n            else -> {\n                // Default action\n            }\n        }\n    }\n}\n```\n\n## ✅ Testing\n\n### Send Test Send test push:\n   ```bash\n   curl -X POST https://notifications-api.visiobook.com/api/v1/push/send \\\n     -H \"X-API-Key: your-api-key\" \\\n     -H \"Content-Type: application/json\" \\\n     -d '{\n       \"userId\": \"your-user-id\",\n       \"title\": \"Test\",\n       \"body\": \"This is a test notification\"\n     }'\n   ```\n\n## 🐛 Troubleshooting\n\n### Token not being generated\n- Check Google Play Services is installed on device\n- Verify `google-services.json` is in the correct location\n- Ensure Firebase project is correctly configured\n\n### Notifications not received\n- Check `POST_NOTIFICATIONS` permission is granted (Android 13+)\n- Verify notification channel is created\n- Check device battery saver isn't blocking notifications\n\n### App crashes\n- Run `./gradlew clean build`\n- Ensure all dependencies are up to date\n- Check logcat for detailed error messages\n\n## 📚 Resources\n\n- [Firebase Android Documentation](https://firebase.google.com/docs/messaging/android/client)\n- [Android Notification Guide](https://developer.android.com/guide/topics/ui/notifiers/notifications)\n- [Google Play Services Setup](https://developers.google.com/android/guides/setup)\n"
