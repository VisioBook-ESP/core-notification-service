# iOS Integration Guide\n\nThis guide shows how to integrate Firebase push notifications in your iOS app.\n\n## 📋 Prerequisites\n\n- Xcode 14+\n- iOS 12.0+\n- CocoaPods\n- Firebase account with iOS app configured\n- APNs certificate uploaded to Firebase\n\n## 🚀 Installation\n\n### 1. Add Firebase to Podfile\n\n```ruby\n# Podfile\nplatform :ios, '12.0'\n\ntarget 'Visiobook' do\n  pod 'Firebase/Messaging', '~> 10.0'\n  pod 'Firebase/Core', '~> 10.0'\nend\n```\n\nInstall pods:\n```bash\npod install\n```\n\n### 2. Download GoogleService-Info.plist\n\n1. Go to [Firebase Console](https://console.firebase.google.com)\n2. Select your project → Project Settings\n3. Go to \"Your apps\" → iOS section\n4. Download `GoogleService-Info.plist`\n5. Drag into Xcode (check \"Copy items if needed\")\n\n## 🔧 Implementation\n\n### Step 1: Configure AppDelegate\n\n```swift\nimport Firebase\nimport UserNotifications\nimport UIKit\n\n@main\nclass AppDelegate: UIResponder, UIApplicationDelegate, UNUserNotificationCenterDelegate {\n    \n    func application(\n        _ application: UIApplication,\n        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?\n    ) -> Bool {\n        \n        // Initialize Firebase\n        FirebaseApp.configure()\n        \n        // Request user permission for notifications\n        UNUserNotificationCenter.current().delegate = self\n        UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge]) { granted, _ in\n            if granted {\n                DispatchQueue.main.async {\n                    UIApplication.shared.registerForRemoteNotifications()\n                }\n            }\n        }\n        \n        // Setup Messaging delegate\n        Messaging.messaging().delegate = self\n        \n        return true\n    }\n    \n    // Handle successful remote notification registration\n    func application(\n        _ application: UIApplication,\n        didRegisterForRemoteNotificationsWithDeviceToken deviceToken: Data\n    ) {\n        // Firebase Messaging will handle this automatically\n        Messaging.messaging().apnsToken = deviceToken\n    }\n    \n    // Handle notification when app is in foreground\n    func userNotificationCenter(\n        _ center: UNUserNotificationCenter,\n        willPresent notification: UNNotification,\n        withCompletionHandler completionHandler: @escaping (UNNotificationPresentationOptions) -> Void\n    ) {\n        let userInfo = notification.request.content.userInfo\n        \n        // Handle foreground notification\n        if let title = userInfo[\"title\"] as? String,\n           let body = userInfo[\"body\"] as? String {\n            print(\"🔔 Foreground notification: \\(title) - \\(body)\")\n        }\n        \n        // Show notification even when app is in foreground\n        completionHandler([.banner, .sound, .badge])\n    }\n    \n    // Handle notification tap\n    func userNotificationCenter(\n        _ center: UNUserNotificationCenter,\n        didReceive response: UNNotificationResponse,\n        withCompletionHandler completionHandler: @escaping () -> Void\n    ) {\n        let userInfo = response.notification.request.content.userInfo\n        handleNotificationTap(userInfo)\n        completionHandler()\n    }\n    \n    private func handleNotificationTap(_ userInfo: [AnyHashable: Any]) {\n        // Navigate based on notification data\n        if let action = userInfo[\"action\"] as? String {\n            print(\"🎯 User tapped notification with action: \\(action)\")\n            switch action {\n            case \"open_message\":\n                if let messageId = userInfo[\"messageId\"] as? String {\n                    // Navigate to message\n                }\n            case \"open_payment\":\n                // Navigate to payments\n                break\n            default:\n                break\n            }\n        }\n    }\n}\n\n// MARK: - Messaging Delegate\nextension AppDelegate: MessagingDelegate {\n    \n    // Called when FCM token is generated or refreshed\n    func messaging(\n        _ messaging: Messaging,\n        didReceiveRegistrationToken fcmToken: String?\n    ) {\n        guard let fcmToken = fcmToken else { return }\n        \n        print(\"🔔 FCM Token: \\(fcmToken)\")\n        \n        // Send this token to your backend\n        registerDeviceWithBackend(token: fcmToken)\n    }\n    \n    private func registerDeviceWithBackend(token: String) {\n        guard let userId = getUserId(),\n              let jwtToken = getJWTToken() else {\n            print(\"❌ User not authenticated\")\n            return\n        }\n        \n        let deviceId = UIDevice.current.identifierForVendor?.uuidString ?? \"unknown\"\n        \n        let url = URL(string: \"https://notifications-api.visiobook.com/api/v1/push/subscribe\")!\n        var request = URLRequest(url: url)\n        request.httpMethod = \"POST\"\n        request.addValue(\"Bearer \\(jwtToken)\", forHTTPHeaderField: \"Authorization\")\n        request.addValue(\"application/json\", forHTTPHeaderField: \"Content-Type\")\n        \n        let body: [String: Any] = [\n            \"deviceId\": deviceId,\n            \"platform\": \"ios\",\n            \"token\": token\n        ]\n        \n        request.httpBody = try? JSONSerialization.data(withJSONObject: body)\n        \n        URLSession.shared.dataTask(with: request) { data, response, error in\n            if let error = error {\n                print(\"❌ Failed to register device: \\(error.localizedDescription)\")\n                return\n            }\n            \n            if let httpResponse = response as? HTTPURLResponse, httpResponse.statusCode == 201 {\n                print(\"✅ Device registered successfully\")\n            }\n        }.resume()\n    }\n    \n    private func getUserId() -> String? {\n        // Get from local storage or auth service\n        return UserDefaults.standard.string(forKey: \"userId\")\n    }\n    \n    private func getJWTToken() -> String? {\n        // Get from Keychain or local storage\n        return UserDefaults.standard.string(forKey: \"jwtToken\")\n    }\n}\n```\n\n### Step 2: Request Permissions\n\n```swift\nimport UserNotifications\n\nclass NotificationManager {\n    static let shared = NotificationManager()\n    \n    func requestNotificationPermission(completion: @escaping (Bool) -> Void) {\n        UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge]) { granted, error in\n            if granted {\n                DispatchQueue.main.async {\n                    UIApplication.shared.registerForRemoteNotifications()\n                }\n            }\n            completion(granted)\n        }\n    }\n    \n    func checkNotificationSettings() {\n        UNUserNotificationCenter.current().getNotificationSettings { settings in\n            DispatchQueue.main.async {\n                switch settings.authorizationStatus {\n                case .authorized:\n                    print(\"✅ Notifications authorized\")\n                case .denied:\n                    print(\"❌ Notifications denied - user must enable in Settings\")\n                case .notDetermined:\n                    print(\"⚠️ Notifications not determined yet\")\n                case .provisional:\n                    print(\"ℹ️ Provisional notifications granted\")\n                case .ephemeral:\n                    print(\"ℹ️ Ephemeral notifications granted\")\n                @unknown default:\n                    break\n                }\n            }\n        }\n    }\n}\n```\n\n### Step 3: Handle Background Notifications\n\n```swift\n// In SceneDelegate or SwiftUI App\n@main\nstruct VisioBookApp: App {\n    @UIApplicationDelegateAdaptor(AppDelegate.self) var appDelegate\n    \n    var body: some Scene {\n        WindowGroup {\n            ContentView()\n                .onReceive(\n                    NotificationCenter.default.publisher(\n                        for: UIApplication.didBecomeActiveNotification\n                    )\n                ) { _ in\n                    // App came to foreground\n                    Messaging.messaging().token { token, error in\n                        if let error = error {\n                            print(\"❌ Error getting FCM token: \\(error)\")\n                        } else if let token = token {\n                            print(\"🔔 FCM Token: \\(token)\")\n                        }\n                    }\n                }\n        }\n    }\n}\n```\n\n## ✅ Testing\n\n### Send Test Notification\n\n1. Get your FCM token from the logs\n2. Use Firebase Console → Cloud Messaging → Send a test message\n3. Or use the backend API:\n\n```bash\ncurl -X POST https://notifications-api.visiobook.com/api/v1/push/send \\\n  -H \"X-API-Key: your-api-key\" \\\n  -H \"Content-Type: application/json\" \\\n  -d '{\n    \"userId\": \"your-user-id\",\n    \"title\": \"Test\",\n    \"body\": \"This is a test notification\"\n  }'\n```\n\n## 🐛 Troubleshooting\n\n### Token not being generated\n- Ensure `GoogleService-Info.plist` is in the project\n- Check that APNs certificate is uploaded to Firebase\n- Verify app is registered with Firebase Console\n\n### Notifications not received\n- Check notification permissions in Settings → Visiobook\n- Verify device is connected and authenticated\n- Check Firebase Console logs\n\n### App crashes on startup\n- Verify Firebase pod versions\n- Run `pod install --repo-update`\n- Clean build folder: ⌘ + Shift + K\n\n## 📚 Resources\n\n- [Firebase Documentation](https://firebase.google.com/docs/messaging/ios/client)\n- [Apple Push Notification Guide](https://developer.apple.com/documentation/usernotifications)\n- [Notification Center API](https://developer.apple.com/documentation/usernotifications/unusernotificationcenter)\n"
